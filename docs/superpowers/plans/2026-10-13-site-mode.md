**Goal:** 为构建工具链新增 `--site` 模式,把 44 章教程构建成每章一页、带侧边栏/翻页/pagefind 中文搜索的静态站点,由现有 GitHub Pages workflow 发布。

/**
 * lib.mjs — 构建共享件(build.mjs 与 site.mjs 共用)。
 * 只放与输出形态无关的渲染件:markdown-it 实例、转义/元数据小工具、
 * vendor 同步、mermaid 页尾脚本。
 */
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

import hljs from 'highlight.js';
import markdownIt from 'markdown-it';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const BUILD_DIR = __dirname;
const ROOT_DIR = path.resolve(BUILD_DIR, '..');

export const md = markdownIt({
  html: true,
  linkify: true,
  highlight(str, lang) {
    if (lang && hljs.getLanguage(lang)) {
      try {
        return hljs.highlight(str, { language: lang }).value;
      } catch {
        /* 高亮失败时退回默认转义 */
      }
    }
    return ''; // 交给 markdown-it 默认转义
  },
});

/** ```mermaid 围栏 → <div class="mermaid">(由页内 mermaid.min.js 渲染为 SVG,其余围栏走默认规则) */
const defaultFence = md.renderer.rules.fence;
md.renderer.rules.fence = (tokens, idx, options, env, self) => {
  const token = tokens[idx];
  if (token.info.trim().split(/\s+/)[0] === 'mermaid') {
    return `<div class="mermaid">\n${token.content}</div>\n`;
  }
  return defaultFence(tokens, idx, options, env, self);
};

/** Windows 路径 → 正斜杠路径(供 file:// URL 与 Edge 参数使用) */
export const toPosix = (p) => p.split(path.sep).join('/');

export function escapeHtml(s) {
  return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}

export function splitFrontMatter(raw) {
  const m = raw.match(/^---\r?\n([\s\S]*?)\r?\n---\r?\n?/);
  if (!m) return { meta: {}, body: raw };
  const meta = {};
  for (const line of m[1].split(/\r?\n/)) {
    const kv = line.match(/^([A-Za-z_][\w-]*):\s*(.*)$/);
    if (kv) meta[kv[1]] = kv[2].trim();
  }
  return { meta, body: raw.slice(m[0].length) };
}

/** 构建时复制到 assets/vendor/ 的第三方脚本,HTML 以相对路径引用(file:// 可直接加载) */
export const VENDOR_SCRIPTS = [
  { from: 'node_modules/mermaid/dist/mermaid.min.js', to: 'assets/vendor/mermaid.min.js' },
  { from: 'node_modules/pagedjs/dist/paged.polyfill.js', to: 'assets/vendor/paged.polyfill.js' },
];

/** 把 vendor 脚本从 node_modules 复制到 assets/vendor/ */
export function syncVendorAssets() {
  for (const v of VENDOR_SCRIPTS) {
    const from = path.join(BUILD_DIR, ...v.from.split('/'));
    const to = path.join(BUILD_DIR, ...v.to.split('/'));
    fs.mkdirSync(path.dirname(to), { recursive: true });
    fs.copyFileSync(from, to);
  }
}

/**
 * 无 Paged.js 的页尾脚本(站点页与 --pdf-fallback 共用):
 * DOMContentLoaded 后渲染全部 .mermaid 为 SVG,完成置 window.__BOOK_READY=true。
 * @param {string} vendorPrefix mermaid.min.js 相对当前页面的目录前缀,含尾斜杠
 */
export function renderMermaidScripts(vendorPrefix) {
  return `<script src="${vendorPrefix}mermaid.min.js"></script>
<script>
window.__BOOK_ERRORS=[];
mermaid.initialize({ startOnLoad: false, theme: "neutral", flowchart: { htmlLabels: false } });
document.addEventListener('DOMContentLoaded', () => {
  window.mermaid.run({ querySelector: '.mermaid', suppressErrors: true })
    .catch((err) => { window.__BOOK_ERRORS.push('mermaid: ' + (err && err.message || err)); })
    .finally(() => {
      const failed = document.querySelectorAll('.mermaid:not(:has(svg))').length;
      if (failed) window.__BOOK_ERRORS.push('mermaid: ' + failed + ' 张图未渲染为 SVG');
      window.__BOOK_READY = true;
    });
});
</script>`;
}
```

/**
 * site.mjs — --site 模式的静态站点构建器。
 * 纯函数(本文件上半部)可单测;buildSite()(Task 4)负责文件总装。
 */
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

import { BOOK, PARTS } from './manifest.mjs';
import { escapeHtml, md, renderMermaidScripts, splitFrontMatter } from './lib.mjs';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const BUILD_DIR = __dirname;
const ROOT_DIR = path.resolve(BUILD_DIR, '..');
const SITE_DIR = path.join(ROOT_DIR, 'output', 'site');
const CHAPTERS_DIR = path.join(ROOT_DIR, 'chapters');

/** demo 目录链接在站点内无对应页面,统一降级到 GitHub 源文件 */
const GITHUB_BASE = 'https://github.com/ybak/agentscope-2.0-tutorial/blob/main/';

/**
 * 站点链接重写:相对 .md 链接 → .html(保留锚点);
 * ../demo/…md → GitHub blob 地址;其余 ../ 与外链原样。
 */
export function rewriteMarkdownLinks(html) {
  return html.replace(/href="([^"]+\.md)(#[^"]*)?"/g, (full, target, hash = '') => {
    if (target.startsWith('../demo/')) {
      return `href="${GITHUB_BASE}${target.replace(/^\.\.\//, '')}${hash}"`;
    }
    if (target.startsWith('../')) return full;
    return `href="${target.replace(/\.md$/, '.html')}${hash}"`;
  });
}

const toPageHref = (mdFile) => `chapters/${mdFile.replace(/\.md$/, '.html')}`;

/**
 * 全站页面扁平序列(上一篇/下一篇依据):序言 → 各篇(篇导读 → 各章)→ 附录。
 * part-0 的 intro 即序言章,只在章位置登记一次(kind 仍为 part-intro)。
 */
export function flattenPages(parts) {
  const pages = [];
  for (const part of parts) {
    if (part.intro && !part.chapters.some((c) => c.file === part.intro)) {
      pages.push({ href: toPageHref(part.intro), title: part.title, part: part.title, kind: 'part-intro' });
    }
    for (const ch of part.chapters) {
      const kind = part.intro && ch.file === part.intro ? 'part-intro'
        : ch.chapter != null ? 'chapter' : 'extra';
      pages.push({ href: toPageHref(ch.file), title: ch.title, part: part.title, kind });
    }
  }
  return pages;
}
```

```js
// ---------- 渲染层 ----------

const readBuildAsset = (...seg) => fs.readFileSync(path.join(BUILD_DIR, ...seg), 'utf8');

/**
 * 整页外壳。rootPrefix:站点根相对前缀(章节页 '../',根页面 '');
 * searchUi:引 pagefind UI 脚本(仅搜索页);needMermaid:引 mermaid(仅章内含图页面需要,统一开启也无害,此处按页关闭以省请求)。
 */
export function renderShell({ title, content, rootPrefix, sidebar = null, needMermaid = false, searchUi = false }) {
  const style = readBuildAsset('assets', 'style.css');
  const siteStyle = readBuildAsset('assets', 'site.css');
  const hljsTheme = readBuildAsset('node_modules', 'highlight.js', 'styles', 'github.css');
  const vendor = `${rootPrefix}build/assets/vendor/`;
  const tail = [
    needMermaid ? renderMermaidScripts(vendor) : '',
    searchUi ? `<script src="${vendor}pagefind-ui.js"></script>
<script>
window.addEventListener('DOMContentLoaded', () => {
  new PagefindUI({ element: '#search', showImages: false });
  const q = new URLSearchParams(location.search).get('q');
  if (q) {
    const input = document.querySelector('#search input');
    input.value = q;
    input.dispatchEvent(new Event('input', { bubbles: true }));
  }
});
</script>` : '',
  ].join('\n');
  return `<!DOCTYPE html>
<html lang="zh-CN">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<title>${escapeHtml(title)}${title === BOOK.title ? '' : ` · ${escapeHtml(BOOK.title)}`}</title>
<style>
${style}
</style>
<style>
${hljsTheme}
</style>
<style>
${siteStyle}
</style>
</head>
<body>
<header class="site-header">
<button class="nav-toggle" aria-label="打开目录">☰</button>
<a class="site-title" href="${rootPrefix}index.html">${escapeHtml(BOOK.title)}</a>
<form class="site-search" action="${rootPrefix}search.html"><input type="search" name="q" placeholder="搜索全书…"></form>
</header>
<div class="site-body">
${sidebar ? `<aside class="site-aside"><button class="nav-toggle" aria-label="关闭目录" onclick="document.querySelector('.site-aside').classList.remove('open')">✕</button>${sidebar}</aside>` : ''}
<main class="site-main">
${content}
</main>
</div>
${tail}
<script>
document.querySelector('.site-header .nav-toggle')?.addEventListener('click',
  () => document.querySelector('.site-aside')?.classList.toggle('open'));
</script>
</body>
</html>
`;
}

/** 左侧两级目录:篇 = <details>(当前篇 open,标题链到篇导读页;无 intro 的篇为纯文本),章 = <a>(当前章高亮) */
export function renderSidebar(parts, currentHref) {
  const items = parts.map((part) => {
    const introHref = part.intro ? toPageHref(part.intro) : null;
    const partActive = introHref != null && introHref === currentHref;
    const isOpen = partActive || part.chapters.some((c) => toPageHref(c.file) === currentHref);
    const links = part.chapters.map((c) => {
      const href = toPageHref(c.file);
      const active = href === currentHref ? ' class="active"' : '';
      const label = c.chapter != null ? `${c.chapter}. ${c.title}` : c.title;
      return `<a href="../${href}"${active}>${escapeHtml(label)}</a>`;
    }).join('');
    const summary = introHref
      ? `<a href="../${introHref}"${partActive ? ' class="active"' : ''}>${escapeHtml(part.title)}</a>`
      : escapeHtml(part.title);
    return `<details${isOpen ? ' open' : ''}><summary>${summary}</summary>${links}</details>`;
  }).join('');
  return `<nav class="site-nav">${items}</nav>`;
}

/** 上一篇/下一篇;链接带 ../ 前缀(仅章节页使用) */
export function renderPager(flat, idx) {
  const link = (p, label) => `<a class="pager-link" href="../${p.href}"><span class="pager-label">${label}</span><span class="pager-title">${escapeHtml(p.title)}</span></a>`;
  const prev = idx > 0 ? link(flat[idx - 1], '← 上一篇') : '<span></span>';
  const next = idx < flat.length - 1 ? link(flat[idx + 1], '下一篇 →') : '<span></span>';
  return `<nav class="pager">${prev}${next}</nav>`;
}

/** 首页:封面信息 + 七篇卡片(篇标题 → 导读页,其下章链接)。rootPrefix 为 ''(根页面) */
export function renderIndexPage(pages) {
  const byPart = new Map();
  for (const p of pages) {
    if (!byPart.has(p.part)) byPart.set(p.part, []);
    byPart.get(p.part).push(p);
  }
  const cards = [...byPart].map(([partTitle, items]) => `<section class="home-card">
<h3><a href="${items[0].href}">${escapeHtml(partTitle)}</a></h3>
${items.map((p) => `<a href="${p.href}">${escapeHtml(p.title)}</a>`).join('\n')}
</section>`).join('\n');
  const content = `<section class="cover">
<h1 class="cover-title">${escapeHtml(BOOK.title)}</h1>
<p class="cover-subtitle">${escapeHtml(BOOK.subtitle)}</p>
<hr class="cover-rule">
<p class="cover-note">基于 AgentScope Java v2 官方文档整合 · 在线阅读版</p>
<p class="cover-date">${escapeHtml(BOOK.date)}</p>
</section>
<div class="home-cards">
${cards}
</div>`;
  return renderShell({ title: BOOK.title, content, rootPrefix: '' });
}

/** 搜索页:pagefind UI 挂载点;data-pagefind-ignore 防止自身入索引 */
export function renderSearchPage() {
  return renderShell({
    title: '搜索',
    content: '<h1>搜索</h1>\n<div id="search"></div>',
    rootPrefix: '',
    searchUi: true,
  }).replace('<body>', '<body data-pagefind-ignore>');
}

// ---------- 总装 ----------

/** 章节页渲染:正文(含链接重写) + 侧边栏 + 翻页 + mermaid */
function renderChapterPage(body, currentHref, flat, idx) {
  return renderShell({
    title: flat[idx].title,
    content: rewriteMarkdownLinks(md.render(body)) + renderPager(flat, idx),
    rootPrefix: '../',
    sidebar: renderSidebar(PARTS, currentHref),
    needMermaid: true,
  });
}

/** 构建期校验:残留 .md 链接与缺失图片 → [site][warn](不中断) */
function auditPage(html, siteDir, label, warns) {
  for (const m of html.matchAll(/href="[^"]*\.md[#"]/g)) {
    warns.push(`[site][warn] ${label}: 残留 .md 链接 ${m[0]}`);
  }
  for (const m of html.matchAll(/src="(\.\.\/assets\/[^"]+)"/g)) {
    if (!fs.existsSync(path.join(siteDir, 'chapters', m[1]))) {
      warns.push(`[site][warn] ${label}: 缺失图片 ${m[1]}`);
    }
  }
}

/**
 * 构建 output/site/:镜像仓库目录(chapters/ + assets/ + build/assets/vendor/),
 * 章节 ../assets/images/ 引用零改动可解析。返回站点目录绝对路径。
 */
export function buildSite() {
  const warns = [];
  fs.rmSync(SITE_DIR, { recursive: true, force: true });
  fs.mkdirSync(path.join(SITE_DIR, 'chapters'), { recursive: true });

  // 资源:仓库 assets 原样 + vendor(mermaid 经 syncVendorAssets;pagefind UI 从 node_modules)
  fs.cpSync(path.join(ROOT_DIR, 'assets'), path.join(SITE_DIR, 'assets'), { recursive: true });
  syncVendorAssets();
  fs.mkdirSync(path.join(SITE_DIR, 'build', 'assets', 'vendor'), { recursive: true });
  fs.copyFileSync(path.join(BUILD_DIR, 'assets', 'vendor', 'mermaid.min.js'),
    path.join(SITE_DIR, 'build', 'assets', 'vendor', 'mermaid.min.js'));
  const uiDist = path.join(BUILD_DIR, 'node_modules', '@pagefind', 'default-ui', 'dist');
  for (const f of ['pagefind-ui.js', 'pagefind-ui.css']) {
    fs.copyFileSync(path.join(uiDist, f), path.join(SITE_DIR, 'build', 'assets', 'vendor', f));
  }

  // 章节/篇导读页(顺序即 flattenPages,manifest 中缺文件 → warn 跳过)
  const flat = flattenPages(PARTS);
  const byHref = new Map();
  for (const part of PARTS) {
    const partPages = [part.intro, ...part.chapters.map((c) => c.file)].filter(Boolean);
    for (const file of partPages) {
      const abs = path.join(CHAPTERS_DIR, file);
      if (!fs.existsSync(abs)) {
        warns.push(`[site][warn] 章节文件缺失,跳过: chapters/${file}`);
        continue;
      }
      byHref.set(toPageHref(file), splitFrontMatter(fs.readFileSync(abs, 'utf8')).body);
    }
  }
  let rendered = 0;
  for (let i = 0; i < flat.length; i++) {
    const body = byHref.get(flat[i].href);
    if (body == null) continue; // 已 warn
    const html = renderChapterPage(body, flat[i].href, flat, i);
    auditPage(html, SITE_DIR, flat[i].href, warns);
    fs.writeFileSync(path.join(SITE_DIR, flat[i].href), html, 'utf8');
    rendered++;
  }

  // 根页面
  fs.writeFileSync(path.join(SITE_DIR, 'index.html'), renderIndexPage(flat), 'utf8');
  fs.writeFileSync(path.join(SITE_DIR, 'search.html'), renderSearchPage(), 'utf8');

  for (const w of warns) console.log(w);
  console.log(`[site] 已渲染 ${rendered}/${flat.length} 页 → ${SITE_DIR}`);
  return SITE_DIR;
}
```
